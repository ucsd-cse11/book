Color averageAround(Raster raster, int col, int row) {
  int r = 0, g = 0, b = 0, total = 0;
  for(int dRow = -1; dRow <= 1; dRow += 1) {
    for(int dCol = -1; dCol <= 1; dCol += 1) {
      if(col + dCol >= raster.width() || col + dCol < 0) { continue; }
      if(row + dRow >= raster.height() || row + dRow < 0) { continue; }
      Color c = raster.colorAt(col + dCol, row + dRow);
      r += c.getRed();
      g += c.getGreen();
      b += c.getBlue();
      total += 1;
    }
  }
  return new Color(r / total, g / total, b / total);
}
Prof: What else needs to change?
Student: That’s it, right? Oh wait now do I change the ranges in the whole blur loop?
Prof: Right, now it can safely blur at [0][0] because it won’t go out of bounds for that row and col.

Prof: This is an example of in-bounds grid processing. It’s pretty common in games and graphical applications. You need special rules for what happens with pixels or grid positions on edges; here we just have fewer to average. In other settings you might treat edges as walls for collisions or make other decisions; the key thing is to not look at out of bounds elements, and pick what the behavior ought to be around the border.

Color averageAround(Raster raster, int col, int row) {
  int r = 0, g = 0, b = 0, total = 0;
  for(int dRow = -1; dRow <= 1; dRow += 1) {
    for(int dCol = -1; dCol <= 1; dCol += 1) {
      if(!inBounds(raster, col + dCol, row + dRow)) { continue; }
      Color c = raster.colorAt(col + dCol, row + dRow);
      r += c.getRed();
      g += c.getGreen();
      b += c.getBlue();
      total += 1;
    }
  }
  return new Color(r / total, g / total, b / total);
}
Raster boxBlur(Raster raster) {
  Color[][] blurred = new Color[raster.height()][raster.width()];
  for(int row = 0; row < raster.height(); row += 1) {
    for(int col = 0; col < raster.width(); col += 1) {
      blurred[row][col] = averageAround(raster, col, row);
    }
  }
  return new Raster(blurred);
}
Color[][] tiny = {
  { Color.WHITE, Color.WHITE, Color.WHITE, Color.WHITE },
  { Color.WHITE, Color.BLACK, Color.BLACK, Color.WHITE },
  { Color.WHITE, Color.WHITE, Color.WHITE, Color.WHITE }
};
Raster small = new Raster(tiny);
Raster smallBlur = boxBlur(small);
