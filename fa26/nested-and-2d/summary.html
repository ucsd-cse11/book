countWhite counts the pure white pixels in a raster. Which arguments go to colorAt?
import java.awt.Color;
int countWhite(Raster raster) {
int count = 0;
for(int row = 0; row < raster.height(); row += 1) {
for(int col = 0; col < raster.width(); col += 1) {
if(raster.colorAt().equals(Color.WHITE)) { count += 1; }
}
}
return count;
}
Color[][] stripes = {
{ Color.WHITE, Color.RED, Color.WHITE },
{ Color.RED, Color.RED, Color.BLUE }
};
