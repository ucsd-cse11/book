import java.awt.Color;

import java.io.File;

/** A simple picture: a fixed-size grid of colored pixels. */
class Pict {
    private final int width;
    private final int height;
    private final Color[][] pixels; // pixels[row][col]

    Pict(int width, int height, Color background) {
        if (width <= 0 || height <= 0) {
            throw new IllegalArgumentException("Pict dimensions must be positive");
        }
        this.width = width;
        this.height = height;
        this.pixels = new Color[height][width];
        fillRect(0, 0, width, height, background);
    }

    int getWidth()  { return width; }
    int getHeight() { return height; }

    Color getPixel(int x, int y)              { return pixels[y][x]; }
    void  setPixel(int x, int y, Color color) { pixels[y][x] = color; }

    /** Fill the rectangle with top-left corner (x, y) and size w x h, clipped to bounds. */
    void fillRect(int x, int y, int w, int h, Color color) {
        for (int r = Math.max(0, y); r < Math.min(height, y + h); r++) {
            for (int c = Math.max(0, x); c < Math.min(width, x + w); c++) {
                pixels[r][c] = color;
            }
        }
    }

    /** Save this picture as a PNG file. */
    void save(String filename) throws IOException {
        BufferedImage img = new BufferedImage(width, height, BufferedImage.TYPE_INT_RGB);
        for (int r = 0; r < height; r++) {
            for (int c = 0; c < width; c++) {
                img.setRGB(c, r, pixels[r][c].getRGB());
            }
        }
        ImageIO.write(img, "png", new File(filename));
    }
}

public class GermanFlag {
    // Official German flag colors
    static final Color BLACK = new Color(0x00, 0x00, 0x00);
    static final Color RED   = new Color(0xDD, 0x00, 0x00);
    static final Color GOLD  = new Color(0xFF, 0xCE, 0x00);

    /** Three equal horizontal stripes (black, red, gold) at the official 3:5 ratio. */
    static Pict makeGermanFlag(int width) {
        int height = width * 3 / 5;
        int stripe = height / 3;
        Pict flag = new Pict(width, height, BLACK);
        flag.fillRect(0, 0,          width, stripe,              BLACK);
        flag.fillRect(0, stripe,     width, stripe,              RED);
        flag.fillRect(0, 2 * stripe, width, height - 2 * stripe, GOLD); // absorbs rounding
        return flag;
    }

    public static void main(String[] args) throws IOException {
        Pict flag = makeGermanFlag(500);
        flag.save("german_flag.png");
        System.out.println("Wrote german_flag.png (" + flag.getWidth() + "x" + flag.getHeight() + ")");
    }
}
