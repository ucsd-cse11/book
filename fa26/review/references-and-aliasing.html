record Menu(int[] pricesCents) {
  int[] discounted() {
    int[] result = new int[pricesCents.length];
    int index = 0;
    for(int p: pricesCents) {
      result[index] = p * 90 / 100;
      index += 1;
    }
    return result;
  }
  void discountInPlace() {
    int index = 0;
    for(int p: pricesCents) {
      pricesCents[index] = p * 90 / 100;
      index += 1;
    }
  }
}
int[] regular = { 2000, 1000 };
Menu lunch = new Menu(regular);
Menu dinner = new Menu(lunch.discounted());
void main() {
  lunch.discountInPlace();
}

What is q.likes when main finishes?

class Post {
  String user;
  String message;
  int likes;
  Post(String user, String message, int likes) {
    this.user = user;
    this.message = message;
    this.likes = likes;
  }
  void like() {
    this.likes = likes + 1;
  }
}
void boost(Post post) {
  post.like();
  post.like();
}
Post p = new Post("joe", "CSE-PACE next Tuesday!", 22);
Post q = p;
void main() {
  boost(q);
  p.like();
}
  1. 25
  2. 24
    Post q = p; copies the reference, not the object, so p.like() changes the one Post that q also names.
  3. 22
    boost’s post holds the same reference as q. Its two calls to like change the one Post there is.
  4. 23
    All three calls to like reach the same object: two through post, one through p.
